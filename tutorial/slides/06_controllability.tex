\begin{frame}{The Concept}

\begin{columns}[c]
\begin{column}{0.45\textwidth}
\begin{itemize}
  \item \textbf{\color{ubaBlue}Controllable} --- actions the controller \emph{decides}:
    extend, retract, grip, release.
  \item \textbf{\color{red!70!black}Uncontrollable} --- environment events that
    \emph{cannot be blocked}: object\_detected, obstacle\_detected.
\end{itemize}

\begin{frame}[fragile]{Controllable Reachability: Return Home}

Can the controller \emph{force} returning to \textbf{Home} from every state?

\lstinputlisting[style=ctxdsl, firstline=95, lastline=102]{../examples/06_controllability.ctxdsl}

\medskip
\begin{itemize}
  \item \ctxdsl{[ (ctrl = controllable) ] X} ensures:
    \begin{itemize}
      \item All uncontrollable transitions stay in the winning region.
      \item At least one controllable transition progresses toward Home.
    \end{itemize}
  \item \textbf{Result:} 4/5 states satisfy --- \textbf{Extended} is excluded
        because \ctxdsl{idle} is uncontrollable there, letting the
        environment stall indefinitely.
  \item The controller cannot force progress from Extended when the
        environment can choose to idle forever.
\end{itemize}

\end{frame}

\vspace{0.5em}
\begin{itemize}
  \item \ctxdsl{source} --- the automaton to restrict.
  \item \ctxdsl{satisfying} --- the property the controller must enforce.
  \item The synthesizer removes \emph{controllable} transitions that would
        violate the property, while preserving all uncontrollable transitions.
\end{itemize}

\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Synthesis Result and Key Rules}

\textbf{Result:} the controller retains all 5 states --- the property is
\alert{realizable} without removing any state.

\vspace{1em}
\textbf{How synthesis works:}
\begin{enumerate}
  \item Compute the \emph{winning region} --- states from which the property
        can be enforced despite the environment.
  \item \emph{Restrict} controllable transitions that leave the winning region.
  \item \emph{Preserve} all uncontrollable transitions (environment cannot be blocked).
\end{enumerate}

\vspace{1em}
\textbf{Key rule for composition:}
\begin{center}
\fbox{\parbox{0.75\textwidth}{%
  Controllable labels must \textbf{not} be shared between automata
  in a composition --- each controller owns its own actions.
}}
\end{center}
