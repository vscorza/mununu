% code.tex — CTXDSL and shell listing styles for Mununu tutorial
% Usage: \input{common/code} (loaded via preamble.tex)

% --- UTF-8 support for listings (em-dash and other non-ASCII in comments) ---
\lstset{
  extendedchars=true,
  inputencoding=utf8,
  literate={—}{{---}}3 {→}{{$\to$}}1 {ε}{{$\varepsilon$}}1
           {≥}{{$\geq$}}1 {≤}{{$\leq$}}1 {≠}{{$\neq$}}1
           {⟨}{{$\langle$}}1 {⟩}{{$\rangle$}}1
           {μ}{{$\mu$}}1 {ν}{{$\nu$}}1,
}

% ---------------------------------------------------------------
% Language: CTXDSL (Mununu DSL for compositional labeled transition systems)
% ---------------------------------------------------------------
\lstdefinelanguage{CTXDSL}{
  morekeywords=[1]{
    context, alphabet, label, constants, const, ranges, range,
    automata, automaton, meta, id, comment,
    controllable, internal,
    states, state, initial, state_groups, group, wildcard,
    transitions, transition, on, epsilon,
    guard, effects,
    variables, var, predicates, predicate,
    parameters, param, in,
    composition, synchronous, asynchronous, superset, members,
    mu_formulas, formula, over, all, body,
    controllers, controller, source, satisfying, export,
    minimize, diagnostics, counterexample,
    deadlock_traces, max_counter_traces, proof_obligations,
  },
  morekeywords=[2]{
    nu, mu, ltl, true, false, i64, bool,
  },
  morecomment=[l]{//},
  morestring=[b]",
  sensitive=true,
}

\lstdefinestyle{ctxdsl}{
  language=CTXDSL,
  basicstyle=\ttfamily\footnotesize,
  keywordstyle=[1]{\color{ubaBlue}\bfseries},
  keywordstyle=[2]{\color{ubaLightBlue}\bfseries},
  commentstyle=\color{ubaGray}\itshape,
  stringstyle=\color{ubaGray},
  numbers=left,
  numberstyle=\tiny\color{ubaGray},
  numbersep=6pt,
  frame=l,
  framerule=1.5pt,
  rulecolor=\color{ubaBlue},
  breaklines=true,
  columns=fullflexible,
  showstringspaces=false,
  tabsize=4,
  xleftmargin=1.2em,
}

% ---------------------------------------------------------------
% LTL / Mu-calculus formula style (for inline formulas)
% ---------------------------------------------------------------
\lstdefinestyle{formula}{
  language=CTXDSL,
  basicstyle=\ttfamily\footnotesize,
  keywordstyle=[1]{\color{ubaBlue}\bfseries},
  keywordstyle=[2]{\color{ubaLightBlue}\bfseries},
  breaklines=true,
  columns=fullflexible,
  frame=none,
}

% ---------------------------------------------------------------
% Style: Shell / terminal output
% ---------------------------------------------------------------
\lstdefinestyle{shell}{
  basicstyle=\ttfamily\footnotesize,
  breaklines=true,
  columns=fullflexible,
  frame=l,
  framerule=1.5pt,
  rulecolor=\color{ubaGray},
  showstringspaces=false,
  xleftmargin=1.2em,
}

% ---------------------------------------------------------------
% Environments (convenient wrappers)
% ---------------------------------------------------------------
\lstnewenvironment{ctxdslcode}[1][]{%
  \lstset{style=ctxdsl, #1}%
}{}

\lstnewenvironment{shellcode}[1][]{%
  \lstset{style=shell, #1}%
}{}

% ---------------------------------------------------------------
% Inline commands
% ---------------------------------------------------------------
\newcommand{\ctxdsl}[1]{\lstinline[style=ctxdsl]{#1}}
\newcommand{\cmd}[1]{{\ttfamily\color{ubaBlue}#1}}
