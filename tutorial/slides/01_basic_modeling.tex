\begin{frame}{What is Mununu?}
  Mununu is a \alert{formal verification tool} for reactive systems modeled as
  Compositional Labeled Transition Systems (CLTS).

  \bigskip
  \begin{itemize}
    \item Model systems as finite-state machines with labeled transitions
    \item Verify properties using mu-calculus and LTL
    \item Synthesize controllers that enforce specifications
    \item Compose subsystems with synchronous or asynchronous semantics
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Why Model with State Machines?}
  Real systems are complex. State machines let us \alert{abstract away}
  implementation details and focus on \textbf{behavior}.

  \bigskip
  \begin{center}
  \begin{tabularx}{0.88\textwidth}{lX}
    \toprule
    \textbf{Benefit} & \textbf{Description} \\
    \midrule
    Succinctness  & Capture protocol logic in a handful of states \\
    Precision     & Every transition is explicit --- no hidden behavior \\
    Composability & Build large systems from small, verified components \\
    Verifiability & Automatically check safety, liveness, reachability \\
    \bottomrule
  \end{tabularx}
  \end{center}
\end{frame}

\begin{frame}[fragile]{CTXDSL: Context and Alphabet}
  Every \texttt{.ctxdsl} file starts with a \ctxdsl{context} block.
  The \ctxdsl{alphabet} declares every event the system can observe.

  \bigskip
  \lstinputlisting[style=ctxdsl, firstline=16, lastline=23]{../examples/01_basic_modeling.ctxdsl}

  \bigskip
  \begin{itemize}
    \item \ctxdsl{context} --- top-level container (one per file)
    \item \ctxdsl{label} --- declares a unique event name
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}[fragile]{CTXDSL: Automaton and States}
  Inside \ctxdsl{automata}, each \ctxdsl{automaton} defines a finite-state machine.

  \bigskip
  \lstinputlisting[style=ctxdsl, firstline=25, lastline=33]{../examples/01_basic_modeling.ctxdsl}

  \bigskip
  \begin{itemize}
    \item \ctxdsl{automaton Name} --- names the FSM
    \item \ctxdsl{state Name initial} --- marks the starting state
    \item \ctxdsl{state Name} --- additional states
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}[fragile]{CTXDSL: Transitions (Unlock and Relock)}
  Transitions connect states via labeled events.

  \bigskip
  \lstinputlisting[style=ctxdsl, firstline=35, lastline=43]{../examples/01_basic_modeling.ctxdsl}

  \bigskip
  \begin{itemize}
    \item \ctxdsl{transition A -> B on label L} --- edge from A to B on event L
    \item Multiple transitions can share the same label (e.g., \texttt{unlock})
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{CTXDSL: Formulas --- Safety and Reachability}

  \begin{itemize}
    \item $\gfp X.\;\Boxx X$ --- \textbf{safety}: every reachable state is valid
    \item $\lfp X.\;(p \lor \Dia X)$ --- \textbf{reachability}: state $p$ can be reached
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}[fragile]{CTXDSL: Formulas --- Recovery and Cycles}
  \lstinputlisting[style=ctxdsl, firstline=69, lastline=80]{../examples/01_basic_modeling.ctxdsl}

  \bigskip
  \begin{itemize}
    \item \textbf{Recovery}: from Lockout, Locked is always reachable
    \item \textbf{Cycle}: from any state, Unlocked is reachable
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}[fragile]{Running It}
  \textbf{Evaluate} a formula:
  \begin{shellcode}
$ mununu context eval 01_basic_modeling.ctxdsl \
    --formula lockout_reachable --automaton DoorLock
Formula lockout_reachable: 5/5 states satisfying
  \end{shellcode}

  \medskip
  \textbf{Summarize} the context:
  \begin{shellcode}
$ mununu context summarize 01_basic_modeling.ctxdsl
Context door_lock: 1 automaton, 4 formulas
  \end{shellcode}
\end{frame}

\begin{frame}{Syntax Summary (1/2)}
  \begin{center}
  \begin{tabularx}{0.88\textwidth}{lX}
    \toprule
    \textbf{Keyword}   & \textbf{Purpose} \\
    \midrule
    \ctxdsl{context}    & Top-level container (one per file) \\
    \ctxdsl{alphabet}   & Signal/event declarations \\
    \ctxdsl{label}      & Declares one event name \\
    \ctxdsl{automata}   & Block containing automaton definitions \\
    \ctxdsl{automaton}  & Defines one finite-state machine \\
    \ctxdsl{states}     & Block listing state declarations \\
    \ctxdsl{state}      & Declares a state; \ctxdsl{initial} marks the start \\
    \bottomrule
  \end{tabularx}
  \end{center}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Syntax Summary (2/2)}
  \begin{center}
  \begin{tabularx}{0.88\textwidth}{lX}
    \toprule
    \textbf{Keyword}    & \textbf{Purpose} \\
    \midrule
    \ctxdsl{transitions} & Block listing transition rules \\
    \ctxdsl{transition}  & One edge: \ctxdsl{A -> B on label L} \\
    \ctxdsl{on}          & Precedes the triggering label(s) \\
    \ctxdsl{mu\_formulas} & Block of property specifications \\
    \ctxdsl{formula}     & One named property \\
    \ctxdsl{over}        & Target automaton for evaluation \\
    \ctxdsl{body}        & The mu-calculus expression \\
    \bottomrule
  \end{tabularx}
  \end{center}
\end{frame}
