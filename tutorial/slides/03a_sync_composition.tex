% ------------------------------------------------------------------
\begin{frame}{Lock-Step Semantics}
  \begin{center}
  \begin{tabularx}{0.88\textwidth}{lX}
    \toprule
    \textbf{Rule} & \textbf{Description} \\
    \midrule
    Shared labels &
      A label present in \emph{both} alphabets (e.g.\ \texttt{transfer})
      must fire in both automata simultaneously. \\
    \addlinespace
    Non-shared labels &
      Labels unique to one automaton (e.g.\ \texttt{produce}, \texttt{consume})
      combine freely with any transition of the other side. \\
    \addlinespace
    Skip labels &
      Each automaton needs its own skip (\texttt{p\_skip}, \texttt{c\_skip})
      so it can ``do nothing'' while the other acts.
      A shared \texttt{idle} would force both to idle together. \\
    \bottomrule
  \end{tabularx}
  \end{center}
\end{frame}

    \node[state]           (DP)  [right=of EP]   {DataReady\\Processing};

    \path
      (EW) edge              node[above] {\tiny produce + c\_skip} (DW)
      (DW) edge              node[right] {\tiny transfer}          (EP)
      (EP) edge              node[below] {\tiny produce + consume} (DW)
      (EP) edge [loop left]  node[left]  {\tiny p\_skip + consume} ()
      (EW) edge [loop left]  node[left]  {\tiny p\_skip + c\_skip} ()
      (DW) edge [loop right] node[right] {\tiny p\_skip + c\_skip} ()
      (EP) edge              node[left]  {\tiny p\_skip + consume} (EW)
      (DP) edge              node[right] {\tiny transfer}          (EP)
      (DW) edge              node[above, sloped] {\tiny produce + c\_skip} (DP)
      (DP) edge [loop right] node[right] {\tiny p\_skip + c\_skip} ()
      ;
  \end{tikzpicture}
  \end{center}
\end{frame}

\begin{frame}[fragile]{Formulas Over Compositions}
  \lstinputlisting[style=ctxdsl, firstline=66, lastline=94]{../examples/03a_sync_composition.ctxdsl}

  \medskip
  \begin{itemize}
    \item \textbf{Structural} formulas (\ctxdsl{over pipeline}) use no state names --- they work on the product.
    \item \textbf{Component} formulas (\ctxdsl{over Producer}) can reference individual state names.
  \end{itemize}
\end{frame}

\begin{frame}{New Syntax Summary}
  \begin{center}
  \begin{tabularx}{0.85\textwidth}{lX}
    \toprule
    \textbf{Keyword}       & \textbf{Purpose} \\
    \midrule
    \ctxdsl{composition}    & Block declaring composed systems \\
    \ctxdsl{synchronous}    & Lock-step composition mode \\
    \ctxdsl{members}        & Lists the automata to compose \\
    \bottomrule
  \end{tabularx}
  \end{center}

  \bigskip
  \textbf{Key takeaway:} Synchronous composition models systems where every component takes a step on every clock cycle. Shared labels enforce coordination; private skip labels allow independent idling.
\end{frame}
