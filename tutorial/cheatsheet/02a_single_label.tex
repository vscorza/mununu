\cheatheader{02a --- Single-Label Transitions}{Controllable Labels and Controllers}

\begin{multicols}{3}
\footnotesize

%% ---- COLUMN 1 ----

\begin{cheatbox}{Controllable Labels}
Labels the controller is allowed to enable/disable:
\begin{lstlisting}[style=ctxdsl]
automaton SpiMaster {
    controllable {
        label cs_assert;
        label cs_deassert;
        label sclk_tick;
        label mosi_write;
        label done;
    }
    states { ... }
    transitions { ... }
}
\end{lstlisting}
Declared \emph{inside} the automaton, before states.
\end{cheatbox}

\begin{cheatbox}{Uncontrollable Labels}
Any label \textbf{not} listed in the \code{controllable} block is \emph{uncontrollable} --- it represents an environment event the controller cannot prevent.

\smallskip
The controller must tolerate all uncontrollable transitions: it can only restrict which controllable labels fire.
\end{cheatbox}

\begin{tipbox}{Design Guideline}
Start by making all labels controllable.  Move labels to uncontrollable only when they represent genuine environment actions (sensor inputs, external interrupts, etc.).
\end{tipbox}

%% ---- COLUMN 2 ----

\begin{cheatbox}{Controller Declaration}
A controller restricts an automaton to satisfy a formula:
\begin{lstlisting}[style=ctxdsl]
controllers {
    controller spi_safety {
        source SpiMaster;
        satisfying safety_invariant;
    }
}
\end{lstlisting}
\begin{itemize}
\item \code{source} --- the automaton (or composition) to control
\item \code{satisfying} --- the mu-calculus formula to enforce
\end{itemize}
\end{cheatbox}

\begin{cheatbox}{Labeled Modalities}
Target specific labels in formulas:

\smallskip
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
Syntax & Meaning \\
\midrule
\code{<labels=\{a\}> $\varphi$} & \emph{Diamond}: $\exists$ an \code{a}-successor satisfying $\varphi$ \\
\code{[labels=\{a\}] $\varphi$} & \emph{Box}: $\forall$ \code{a}-successors satisfy $\varphi$ \\
\bottomrule
\end{tabularx}

\smallskip
Example --- from Idle, \code{cs\_assert} reaches Selected:
\begin{lstlisting}[style=ctxdsl]
body = (! Idle) ||
    < labels = { cs_assert } > Selected;
\end{lstlisting}
\end{cheatbox}

%% ---- COLUMN 3 ----

\begin{cheatbox}{CLI Synthesis}
\begin{lstlisting}[style=shell]
# Synthesize a controller
mununu context synth file.ctxdsl \
    --formula safety_invariant \
    --automaton SpiMaster

# Evaluate (check without synthesizing)
mununu context eval file.ctxdsl \
    --formula transfer_completes \
    --automaton SpiMaster
\end{lstlisting}
\end{cheatbox}

\begin{tipbox}{Synthesis Output}
\begin{itemize}
\item \textbf{Realizable} --- a controller exists that enforces the property. Output shows the number of states retained.
\item \textbf{Unrealizable} --- no controller can enforce the property under all environment behaviors.
\end{itemize}
\end{tipbox}

\begin{warnbox}{Composition Rule}
Controllable labels must \textbf{not} be shared between automata in a composition.  If two automata both declare a label as controllable, synthesis results are undefined.  Use distinct label names or keep shared labels uncontrollable.
\end{warnbox}

\begin{cheatbox}{Useful Formula Patterns}
\begin{lstlisting}[style=ctxdsl]
// Can always return to Idle
mu X. (Idle || <> X)

// All cs_assert transitions safe
[labels = { cs_assert }] Safe

// Liveness under control
nu X. (mu Y. (Goal || <> Y)) && ([] X)
\end{lstlisting}
\end{cheatbox}
