% ── Column 1 ──────────────────────────────────────

\begin{cheatbox}{Syntax}
\begin{lstlisting}[style=ctxdsl]
asynchronous name {
    members [A, B, C];
}
\end{lstlisting}
Declares an asynchronous composition of automata
\code{A}, \code{B}, and \code{C} under the identifier
\code{name}. All listed members must be defined in the
same context.
\end{cheatbox}

\begin{cheatbox}{Semantics}
\begin{itemize}
  \item Any component can take a step \emph{independently}
        of the others.
  \item Components whose label is not involved remain in
        their current state (interleaving).
  \item Labels that appear in the transition sets of
        \textbf{multiple} members \emph{must synchronize}
        --- all participating automata step together.
  \item No \code{skip} / idle labels are required (unlike
        synchronous composition).
  \item The result is a single CLTS whose states are
        tuples of member states.
\end{itemize}
\end{cheatbox}

\columnbreak

% ── Column 2 ──────────────────────────────────────

\begin{cheatbox}{Sync vs Async --- Comparison}
\renewcommand{\arraystretch}{1.15}
\begin{tabularx}{\linewidth}{@{}l X X@{}}
\toprule
\scriptsize\textbf{Aspect} & \scriptsize\textbf{Synchronous} & \scriptsize\textbf{Asynchronous} \\
\midrule
Stepping   & Lock-step (all advance) & Interleaved (one advances) \\
\code{skip} labels & Required for idle members & Not needed \\
Shared labels & Always sync & Sync only when shared \\
State space & Product of states & Full product of states \\
Use case & Tightly coupled & Loosely coupled \\
\bottomrule
\end{tabularx}
\end{cheatbox}

\begin{tipbox}{When to Use Async}
\begin{itemize}
  \item Independent subsystems (e.g.\ multiple sensors).
  \item Components with different clock domains.
  \item Event-driven systems where components react to
        stimuli at their own pace.
  \item When you want to avoid the overhead of explicit
        \code{skip} transitions in every automaton.
\end{itemize}
\end{tipbox}

\begin{cheatbox}{Shared Labels}
A label is \textbf{shared} if it appears in the transition
sets of two or more member automata.

\begin{itemize}
  \item Shared labels act as \emph{synchronization barriers}:
        all automata that know the label must take it
        simultaneously.
  \item Use \textbf{distinct} label names for actions that
        should remain independent.
  \item Shared labels are the primary mechanism for
        inter-component communication in async mode.
\end{itemize}

\medskip
\textbf{Example:} if \code{A} and \code{B} both have label
\code{sync\_ab}, they must fire it together.  \code{C} is
unaffected and may step independently.
\end{cheatbox}

\begin{cheatbox}{State Space}
The composite state space is the Cartesian product of all
member state sets:

\[
  |S| = |S_A| \times |S_B| \times |S_C| \times \cdots
\]

\textbf{Example:} three automata with 2, 2, and 3 states
yield $2 \times 2 \times 3 = 12$ composite states.

\medskip
\begin{itemize}
  \item Async typically produces more reachable transitions
        than sync (more interleavings).
  \item Use property-based reduction or minimization to
        manage large state spaces.
\end{itemize}
\end{cheatbox}

\end{tipbox}
