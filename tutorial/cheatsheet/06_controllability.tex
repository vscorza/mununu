\cheatheader{06 --- Controllability}{Controllable vs Uncontrollable Actions}

\begin{multicols}{3}
\footnotesize

% ── Column 1 ──────────────────────────────────────

\begin{cheatbox}{Controllable Block}
Declare which labels the controller is allowed to restrict:

\begin{lstlisting}[style=ctxdsl]
automaton Actuator {
    controllable {
        label extend;
        label retract;
    }
    // ...
}
\end{lstlisting}

\begin{itemize}
  \item Only \textbf{controllable} labels can be disabled
        by the synthesized controller.
  \item Every label \emph{not} listed in \code{controllable}
        is \textbf{uncontrollable} by default.
\end{itemize}
\end{cheatbox}

\begin{cheatbox}{Uncontrollable Labels}
Uncontrollable labels represent environment events:

\begin{itemize}
  \item Sensor readings, faults, external requests.
  \item The controller \textbf{cannot prevent} them ---
        they always pass through.
  \item The controller must satisfy the property
        \emph{regardless} of when uncontrollable events
        occur.
\end{itemize}
\end{cheatbox}

\columnbreak

% ── Column 2 ──────────────────────────────────────

\begin{cheatbox}{Controller Synthesis (DSL)}
\begin{lstlisting}[style=ctxdsl]
controller SafeCtrl {
    source Plant;
    satisfying Safety;

    // Optional directives:
    minimize;
    export;
    diagnostics {
        counterexample;
        deadlock_traces;
        max_counter_traces 5;
        proof_obligations;
    }
}
\end{lstlisting}

\renewcommand{\arraystretch}{1.1}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\toprule
\scriptsize\textbf{Directive} & \scriptsize\textbf{Effect} \\
\midrule
\code{source}       & Automaton/composition to control \\
\code{satisfying}   & Formula the controller must enforce \\
\code{minimize}     & Reduce controller states \\
\code{export}       & Write controller CLTS to file \\
\code{diagnostics}  & Enable debug output \\
\bottomrule
\end{tabularx}
\end{cheatbox}

\begin{cheatbox}{CLI}
\begin{lstlisting}[style=shell]
mununu context synth file.ctxdsl \
    --formula Safety \
    --automaton Plant \
    --minimize \
    --counterexample
\end{lstlisting}
\end{cheatbox}

\columnbreak

% ── Column 3 ──────────────────────────────────────

\begin{cheatbox}{Synthesis Results}
\renewcommand{\arraystretch}{1.15}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\toprule
\scriptsize\textbf{Outcome} & \scriptsize\textbf{Meaning} \\
\midrule
Realizable   & A controller exists; states count reported. \\
Unrealizable & No controller can enforce the property under
               all uncontrollable behaviors. \\
\bottomrule
\end{tabularx}

\medskip
When realizable, the output includes:
\begin{itemize}
  \item Number of controller states.
  \item Winning state set (states from which the property
        can be enforced).
  \item Optionally: minimized controller, exported CLTS.
\end{itemize}

When unrealizable, enable \code{diagnostics} to obtain
counterexample traces showing how the environment can
force a violation.
\end{cheatbox}

\begin{tipbox}{Game-Theoretic Formulas}
Game-theoretic reachability:
\begin{lstlisting}[style=ctxdsl]
mu X. (target || [ (ctrl = controllable) ] X)
\end{lstlisting}
Computes the \textbf{controllable attractor} --- states where
the controller can force reaching \code{target}.
\end{tipbox}

\begin{warnbox}{Important Rules}
\begin{enumerate}[leftmargin=*]
  \item \textbf{No overlap:} controllable labels must not
        appear as controllable in multiple automata within
        the same composition.
  \item \textbf{Restriction only:} the controller can only
        \emph{disable} controllable actions --- it cannot
        force them to occur.
  \item \textbf{Pass-through:} uncontrollable events always
        pass through; the controller must tolerate every
        possible sequence of environment actions.
\end{enumerate}
\end{warnbox}
