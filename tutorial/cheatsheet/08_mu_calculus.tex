\begin{cheatbox}{Fixpoint Operators}
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\textbf{Operator} & \textbf{Semantics} \\
\midrule
\code{nu X. phi(X)} & Greatest fixpoint --- starts from \emph{all states} and removes violators downward. Captures \textbf{safety} and \textbf{invariance}. \\
\code{mu X. phi(X)} & Least fixpoint --- starts from \emph{no states} and adds satisfiers upward. Captures \textbf{reachability} and \textbf{liveness}. \\
\bottomrule
\end{tabularx}

\smallskip
Intuition: \code{nu} keeps states that \emph{always} satisfy $\varphi$; \code{mu} finds states that \emph{eventually} satisfy $\varphi$.

Fixpoint variables (\code{X}) are recursive: the formula ``unfolds'' until a stable set is reached.
\end{cheatbox}

\begin{cheatbox}{Modalities}
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\textbf{Syntax} & \textbf{Meaning} \\
\midrule
\code{[] phi} & Box: \emph{all} successors satisfy $\varphi$ \\
\code{<> phi} & Diamond: \emph{some} successor satisfies $\varphi$ \\
\bottomrule
\end{tabularx}

\smallskip
Without labels, modalities quantify over \textbf{all} outgoing transitions from the current state.
\end{cheatbox}

\begin{cheatbox}{Boolean Operators}
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\code{phi \&\& psi} & Conjunction (and) \\
\code{phi || psi} & Disjunction (or) \\
\code{! phi} & Negation (not) \\
\code{true} & Holds in every state \\
\code{false} & Holds in no state \\
\bottomrule
\end{tabularx}
\end{cheatbox}

%% ---- COLUMN 2 ----

\begin{cheatbox}{Labeled Modalities}
Restrict modalities to specific labels:
\begin{lstlisting}[style=ctxdsl]
// All a-or-b transitions lead to phi
[ labels = {a, b} ] phi

// Some a-transition leads to phi
< labels = {a} > phi
\end{lstlisting}

\begin{itemize}
  \item Without \code{labels = \{...\}}: considers \textbf{all} transitions.
  \item With labels: considers \textbf{only} transitions on the specified labels.
  \item Multiple labels in the set act as a union.
\end{itemize}
\end{cheatbox}

\begin{cheatbox}{Nesting and Alternation}
Formulas can nest fixpoints and modalities:
\begin{lstlisting}[style=ctxdsl]
// Alternation: nu outside, mu inside
nu X. (mu Y. (p || <> Y)) && ([] X)
\end{lstlisting}

\begin{itemize}
  \item \textbf{Alternation depth} affects complexity --- deeper nesting is more expensive.
  \item Inner \code{mu} resets each time the outer \code{nu} iterates.
  \item Most practical properties have alternation depth $\leq 2$.
\end{itemize}
\end{cheatbox}

%% ---- COLUMN 3 ----

\begin{cheatbox}{Common Patterns}
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\textbf{Pattern} & \textbf{Formula} \\
\midrule
Safety invariant & \code{nu X. ([] X)} \\[2pt]
Reachability & \code{mu X. (target || <> X)} \\[2pt]
GF (inf.\ often) & \code{nu X. (mu Y. (p || <> Y)) \&\& ([] X)} \\[2pt]
No deadlock & \code{nu X. (<> true) \&\& ([] X)} \\[2pt]
Labeled check & \code{(! src) || [ labels = \{a\} ] dst} \\[2pt]
Mutual exclusion & \code{(! A) || (! B)} \\[2pt]
Recovery & \code{(! bad) || (mu Y. (good || <> Y))} \\[2pt]
Ctrl reachability & \code{mu X. (p || [ (ctrl = controllable) ] X)} \\
\bottomrule
\end{tabularx}
\end{cheatbox}

\begin{tipbox}{Reading Patterns}
\begin{itemize}
  \item \textbf{Safety} = ``always stays in good states'' $\Rightarrow$ \code{nu} + \code{[]}
  \item \textbf{Reachability} = ``eventually reaches target'' $\Rightarrow$ \code{mu} + \code{<>}
  \item \textbf{Liveness} = ``always eventually'' $\Rightarrow$ \code{nu} wrapping \code{mu}
  \item Implications (\code{p $\Rightarrow$ q}) encoded as \code{(! p) || q}
\end{itemize}
\end{tipbox}

\begin{warnbox}{Common Mistakes}
\begin{itemize}
  \item Swapping \code{nu}/\code{mu} --- reverses safety vs.\ reachability semantics.
  \item Using \code{<>} (diamond) when \code{[]} (box) is needed --- ``some path'' vs.\ ``all paths''.
  \item Forgetting that \code{[] false} holds vacuously in deadlock states (no successors).
\end{itemize}
\end{warnbox}
